# Homepage (URL launcher) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a new default `/` homepage listing every reachable URL (running stacks' domains + Caddy's manually-configured "extra routes" hosts) as click-to-launch links in a new tab; move the existing stacks list to `/stacks`.

**Architecture:** Pure frontend change, no backend/API changes. Reuses data `Dashboard` already fetches (`stacks`, per-stack `status`, `config.caddy_extra_routes_json`, `https_port`). A small pure function (`buildHomeEntries`) merges the two URL sources into one sorted list; a new `HomePage` component renders it. A `DomainLink` component is extracted out of `StackCard` so both places render the same clickable-domain markup.

**Tech Stack:** React + TypeScript (Vite), Tailwind CSS, `lucide-react` icons. No test runner exists in `frontend/` (repo convention: only `backend/test_*.py` are tested); verification here is `tsc` type-checking + manual click-through, per repo convention and the design spec.

- [ ] **Step 1: Create `HomePage.tsx`**

```tsx
import { PackageOpen } from 'lucide-react'
import { stackUrl, type Stack, type StackState } from '../api'
import { DomainLink } from './DomainLink'
import { StackIcon } from './StackIcon'

interface HomePageProps {
  stacks: Stack[]
  statuses: Record<string, StackState>
  extraRoutesJson: string
  httpsPort: number
}

interface HomeEntry {
  key: string
  name: string | null
  domain: string
  icon: unknown
  url: string
}

export function buildHomeEntries(
  stacks: Stack[],
  statuses: Record<string, StackState>,
  extraRoutesJson: string,
  httpsPort: number,
): HomeEntry[] {
  const entries: HomeEntry[] = []

  for (const stack of stacks) {
    if (stack.error) continue
    if (statuses[stack.name] === 'stopped') continue
    const domain = stack.x_litethaus.domain
    if (typeof domain !== 'string' || !domain) continue
    entries.push({
      key: `stack:${stack.name}`,
      name: stack.name,
      domain,
      icon: stack.x_litethaus.icon,
      url: stackUrl(domain, httpsPort),
    })
  }

  for (const domain of extractExtraRouteHosts(extraRoutesJson)) {
    entries.push({ key: `route:${domain}`, name: null, domain, icon: null, url: stackUrl(domain, httpsPort) })
  }

  entries.sort((a, b) => (a.name ?? a.domain).localeCompare(b.name ?? b.domain))
  return entries
}

// Extracts every `host` string out of a caddy_extra_routes_json array, i.e.
// every route shaped like {"match": [{"host": ["foo.example.com"]}], ...}.
// Malformed JSON (or JSON that isn't an array) yields no hosts, same
// silent-skip behavior as caddy_service.py's own fallback server-side.
function extractExtraRouteHosts(extraRoutesJson: string): string[] {
  if (!extraRoutesJson.trim()) return []
  let parsed: unknown
  try {
    parsed = JSON.parse(extraRoutesJson)
  } catch {
    return []
  }
  if (!Array.isArray(parsed)) return []

  const hosts = new Set<string>()
  for (const route of parsed) {
    const matches = (route as { match?: unknown } | null)?.match
    if (!Array.isArray(matches)) continue
    for (const matcher of matches) {
      const hostList = (matcher as { host?: unknown } | null)?.host
      if (!Array.isArray(hostList)) continue
      for (const host of hostList) {
        if (typeof host === 'string' && host) hosts.add(host)
      }
    }
  }
  return [...hosts]
}

export function HomePage({ stacks, statuses, extraRoutesJson, httpsPort }: HomePageProps) {
  const entries = buildHomeEntries(stacks, statuses, extraRoutesJson, httpsPort)

  if (entries.length === 0) {
    return (
      <div className="flex flex-col items-center gap-2 py-16 text-center text-neutral-400 dark:text-neutral-500">
        <PackageOpen size={28} />
        <p className="text-sm">No running stacks or custom routes with a domain configured yet.</p>
      </div>
    )
  }

  return (
    <div className="flex flex-col gap-2">
      {entries.map((entry) => (
        <div
          key={entry.key}
          className="flex items-center gap-3 rounded-lg border border-neutral-200 bg-white px-4 py-2.5 dark:border-neutral-800 dark:bg-neutral-900"
        >
          <StackIcon icon={entry.icon} size={22} />
          {entry.name && <span className="font-medium text-neutral-900 dark:text-neutral-100">{entry.name}</span>}
          <div className="ml-auto">
            <DomainLink href={entry.url} label={entry.domain} />
          </div>
        </div>
      ))}
    </div>
  )
}
```
